71float AnalogIn::ReadVoltage(const int samples) const
72{
73 if (!_valid)
74 {
75 return 0.0f;
76 }
77
78 int accumulatedRaw = 0;
79 int rawValue = 0;
80
81 for (int i = 0; i < samples; ++i)
82 {
83 if (adc_oneshot_read(_handle, _channel, &rawValue) == ESP_OK)
84 {
85 accumulatedRaw += rawValue;
86 }
87 }
88
89 int averageRaw = accumulatedRaw / samples;
90
91 if (_cali_enabled && _cali_handle != nullptr)
92 {
93 int voltageMv = 0;
94 if (adc_cali_raw_to_voltage(_cali_handle, averageRaw, &voltageMv) == ESP_OK)
95 {
96 return static_cast<float>(voltageMv) / 1000.0f;
97 }
98 }
99
100 return (static_cast<float>(averageRaw) * _vref) / 4095.0f;
101}
